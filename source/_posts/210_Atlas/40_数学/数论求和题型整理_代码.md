## 0. 公共实现

### 0.1 线性筛：$\mu,\varphi$

```cpp title:"Linear Sieve" fold
int pri[maxn], pcnt;
bool isp[maxn];
ll phi[maxn];
int mu[maxn];

void sieve( int N ) {
    pcnt = 0;
    fill( isp, isp + N + 1, true );
    isp[0] = isp[1] = false;
    phi[1] = 1, mu[1] = 1;
    for( int i = 2; i <= N; ++ i ) {
        if( isp[i] ) {
            pri[++ pcnt] = i;
            phi[i] = i - 1;
            mu[i] = -1;
        }
        for( int j = 1; j <= pcnt && 1LL * i * pri[j] <= N; ++ j ) {
            int p = pri[j];
            isp[i * p] = false;
            if( i % p == 0 ) {
                phi[i * p] = phi[i] * p;
                mu[i * p] = 0;
                break;
            }
            phi[i * p] = phi[i] * ( p - 1 );
            mu[i * p] = -mu[i];
        }
    }
}
```

```cpp title:"Q6 P2303 Longge 的问题" fold
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const int maxn = 2e6 + 5;

ll read( ) { /* 模板代码略 */ }

ll phi( ll n ) {
    ll res = n;
    for( ll i = 2; i <= n / i; ++ i ) {
        if( n % i == 0 ) {
            res = res / i * ( i - 1 );
            while( n % i == 0 ) n /= i;
        }
    }
    if( n > 1 ) res = res / n * ( n - 1 );
    return res;
}

void solve( ) {
    ll n = read( );
    ll ans = 0;
    for( ll d = 1; d * d <= n; ++ d ) {
        if( n % d != 0 ) continue;
        ans += phi( d ) * n / d;
        if( d * d != n ) ans += phi( n / d ) * d;
    }
    cout << ans << '\n';
}

    int T = 1;
    while( T -- ) solve( );
```

### Q8 P4449 于神之怒加强版

```cpp title:"Q8 P4449 于神之怒加强版" fold
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const int maxn = 5e6 + 5;
const int mod = 1e9 + 7;

ll read( ) { /* 模板代码略 */ }

ll qpow( ll a, ll k, ll mod ) { /* 模板代码略 */ }

ll K;
int pri[maxn], pcnt;
bool isp[maxn];
ll J[maxn], pre_J[maxn];

void sieve( int N ) {
    pcnt = 0;
    fill( isp, isp + N + 1, true );
    isp[0] = isp[1] = false;
    J[1] = 1;
    for( int i = 2; i <= N; ++ i ) {
        if( isp[i] ) {
            pri[++ pcnt] = i;
            ll pk = qpow( i, K, mod );
            J[i] = ( pk - 1 + mod ) % mod;
        }
        for( int j = 1; j <= pcnt && 1LL * i * pri[j] <= N; ++ j ) {
            int p = pri[j];
            isp[i * p] = 0;
            if( i % p == 0 ) {
                ll pk = ( J[p] + 1 ) % mod;
                J[i * p] = ( J[i] * pk ) % mod;
                break;
            }
            J[i * p] = ( J[i] * J[p] ) % mod;
        }
    }
    for( int i = 1; i <= N; ++ i ) {
        pre_J[i] = ( pre_J[i - 1] + J[i] ) % mod;
    }
}

ll calc2( ll n, ll m, ll k = 1 ) {
    if( n > m ) swap( n, m );
    n /= k, m /= k;
    ll res = 0;
    for( ll l = 1, r; l <= n; l = r + 1 ) {
        ll vn = n / l, vm = m / l;
        r = min( n / vn, m / vm );
        ll sum_J = ( pre_J[r] - pre_J[l - 1] + mod ) % mod;
        res = ( res + ( sum_J * vn % mod * vm % mod ) ) % mod;
    }
    return res;
}

void solve( ) {
    int n = read( ), m = read( );
    cout << calc2( n, m ) << '\n';
}

int main( ) {
    int T = read( );
    K = read( );
    sieve( maxn - 1 );
    while( T -- ) solve( );
    return 0;
}
```

### Q10 P3768 简单的数学题

```cpp title:"Q10 P3768 简单的数学题" fold
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

const int maxn = 5e6 + 5;
ll mod, inv2, inv6;

ll read( ) { /* 模板代码略 */ }

ll qpow( ll a, ll k, ll mod ) { /* 模板代码略 */ }

ll S1( ll n ) {
    n %= mod;
    return n * ( n + 1 ) % mod * inv2 % mod;
}

ll S2( ll n ) {
    n %= mod;
    return n * ( n + 1 ) % mod * ( 2 * n + 1 ) % mod * inv6 % mod;
}

ll S3( ll n ) {
    ll s = S1( n );
    return s * s % mod;
}

unordered_map<ll, ll> mp_h;
int pri[maxn / 10], pcnt;
bool isp[maxn];
ll phi[maxn];
int pre_h[maxn];
ll preN;

void sieve( int N ) {
    preN = N, pcnt = 0;
    mp_h.clear( );
    fill( isp, isp + N + 1, true );
    isp[0] = isp[1] = false;
    phi[1] = 1;
    for( int i = 2; i <= N; ++ i ) {
        if( isp[i] ) pri[++ pcnt] = i, phi[i] = i - 1;
        for( int j = 1; j <= pcnt && 1LL * i * pri[j] <= N; ++ j ) {
            int p = pri[j];
            isp[i * p] = 0;
            if( i % p == 0 ) {
                phi[i * p] = phi[i] * p;
                break;
            }
            phi[i * p] = phi[i] * ( p - 1 );
        }
    }
    for( ll i = 1; i <= N; ++ i ) {
        int hi = i * i % mod * phi[i] % mod;
        pre_h[i] = ( pre_h[i - 1] + hi ) % mod;
    }
}

ll sum_h( ll N ) {
    if( N <= preN ) return pre_h[N];
    if( mp_h.count( N ) ) return mp_h[N];
    ll res = S3( N );
    for( ll l = 2, r; l <= N; l = r + 1 ) {
        ll v = N / l;
        r = N / v;
        ll sum_d2 = ( S2( r ) - S2( l - 1 ) + mod ) % mod;
        res = ( res - sum_d2 * sum_h( v ) % mod + mod ) % mod;
    }
    return mp_h[N] = res;
}

ll calc2( ll n, ll m, ll k = 1 ) {
    if( n > m ) swap( n, m );
    n /= k, m /= k;
    ll res = 0;
    for( ll l = 1, r; l <= n; l = r + 1 ) {
        ll vn = n / l, vm = m / l;
        r = min( n / vn, m / vm );
        ll sum = ( sum_h( r ) - sum_h( l - 1 ) + mod ) % mod;
        res = ( res + sum * S1( vn ) % mod * S1( vm ) % mod ) % mod;
    }
    return res;
}

void solve( ) {
    mod = read( );
    ll n = read( );
    sieve( maxn - 1 );
    inv2 = qpow( 2, mod - 2, mod );
    inv6 = qpow( 6, mod - 2, mod );
    cout << calc2( n, n ) << '\n';
}

int main( ) {
    int T = 1;
    while( T -- ) solve( );
    return 0;
}
```

```cpp title:"Q12 P3312 数表" fold
    }
}

struct Query { int n, m, a, id; };

ll calc2( int n, int m ) {
    if( n > m ) swap( n, m );
    ll res = 0, pre = 0;
    for( int l = 1, r; l <= n; l = r + 1 ) {
        int vn = n / l, vm = m / l;
        r = min( n / vn, m / vm );
        ll cur = bit.ask( r );
        ll sum_G = ( cur - pre + mod ) % mod;
        pre = cur;
        ll cnt = 1ll * vn * vm % mod;
        res = ( res + sum_G * cnt ) % mod;
    }
    return res;
}

void solve( ) {
    int Q = read( );
    vector<Query> qs( Q );
    int N = 0;

    for( int i = 0; i < Q; ++ i ) {
        int n = read( ), m = read( ), a = read( );
        if( n > m ) swap( n, m );
        qs[i] = { n, m, a, i };
        N = max( N, n );
    }

    init( N );
    bit.init( N );

    vector<int> ds( N );
    iota( ds.begin( ), ds.end( ), 1 );
    sort( ds.begin( ), ds.end( ), [&]( int x, int y ) {
        return sigma[x] < sigma[y];
    } );

    sort( qs.begin( ), qs.end( ), []( const Query &x, const Query &y ) {
        return x.a < y.a;
    } );

    vector<ll> ans( Q );
    int ptr = 0;

    for( auto [n, m, a, idx] : qs ) {
        while( ptr < N && sigma[ds[ptr]] <= a ) {
            int d = ds[ptr ++];
            for( int k = 1; 1ll * d * k <= N; ++ k ) {
                if( mu[k] == 0 ) continue;
                bit.add( d * k, 1ll * sigma[d] * mu[k] );
            }
        }
        ans[idx] = calc2( n, m );
    }

    for( ll x : ans ) cout << x << '\n';
}

int main( ) {
    int T = 1;
    while( T -- ) solve( );
    return 0;
}
```
