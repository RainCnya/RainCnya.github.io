# 水の数列：阈值激活与段数查询

> [!question] [P5012 水の数列](https://www.luogu.com.cn/problem/P5012)
> 选择阈值 $x$，标记所有 $a_i\le x$ 的位置。设连续标记段的长度平方和为 $val$、段数为 $cnt$，得分为 $val/x$。每次强制在线询问要求 $cnt\in[l,r]$，输出最优方案的 $val$ 和 $x$。
>
> 数据规模：$n\le10^6$，$T\le10^3$，$a_i\le10^6$。

当阈值 $x$ 逐渐增大时，位置只会从未标记变为已标记，不会撤销。于是可以把所有位置按 $a_i$ 排序，从小到大激活，并用并查集维护当前连续标记段。

最开始的预处理方向已经是按权值激活，但实现中把所有位置初始化成了有效并查集节点，导致尚未激活的位置也会参与合并；同时把区间数和平方和当成每个权值组内的局部量，也无法表示当前完整标记状态。

## 激活位置时维护段数与平方和

设当前连续段数量为 $cnt$，长度平方和为 $val$。激活一个新位置时，它先形成长度为 $1$ 的新区间：

$$
cnt\mathrel{+}=1,\qquad val\mathrel{+}=1.
$$

随后只与已经激活的左右邻居合并。若两个连续段长度分别为 $s_1,s_2$，合并后

$$
cnt\mathrel{-}=1,
$$

而平方和的变化为

$$
(s_1+s_2)^2-s_1^2-s_2^2=2s_1s_2.
$$

因此并查集根节点只需维护集合大小，每次成功合并时给 `val` 加上 `2LL*s1*s2`。

相同权值的所有位置必须全部激活后才能保存状态，因为选择某个 $x$ 时，所有 $a_i=x$ 的位置会同时被标记，不存在只激活其中一部分的合法状态。

只枚举数组中出现过的 $x$ 已经足够。在相邻两个不同权值之间，标记状态、$cnt$ 和 $val$ 都不变，而 $val/x$ 随 $x$ 增大只会变小，所以该状态一定取能够产生它的最小阈值。

## 从阈值状态压缩到段数状态

不同阈值可能得到相同的段数。令 `best[k]` 表示恰好产生 $k$ 个连续段时，$val/x$ 最大的状态。比较两个状态不能使用浮点数，而应交叉相乘：

$$
\frac{A.val}{A.x}>\frac{B.val}{B.x}
\iff
A.val\cdot B.x>B.val\cdot A.x.
$$

乘积用 `__int128` 防止溢出；若得分相同，按题意选择更大的 $x$。

这样每次询问就变成：在 `best[l...r]` 中寻找自定义比较规则下的最大状态。该比较确定了固定的全序关系，“取较优者”具有结合律和幂等性，因此可以直接使用 ST 表。

为节省空间，ST 表不保存完整 `State`，只保存 `best[]` 的下标：

```cpp title:"P5012" fold
using namespace std;
using ll = long long;
using ull = unsigned long long;
using i128 = __int128_t;

const int maxn = 1e6 + 5;
const int maxcnt = 5e5 + 5;
const int maxlg = 20;
const int mod = 998244353;
const ll inf = 4e18;

ll read( ) {
    ll x = 0, f = 1;
    char ch = getchar( );
    for( ; !isdigit( ch ); ch = getchar( ) ) if( ch == '-' ) f = -1;
    for( ; isdigit( ch ); ch = getchar( ) ) x = x * 10 + ch - '0';
    return x * f;
}

struct DSU {
/* ======================================== */
int fa[maxn], siz[maxn];
void init( int N ) { for( int i = 1; i <= N; ++ i ) fa[i] = 0, siz[i] = 0; }
int find( int x ) { return fa[x] == x ? x : fa[x] = find( fa[x] ); }
bool merge( int x, int y, ll &val ) {
    int rx = find( x ), ry = find( y );
    if( rx == ry ) return 0;
    val += 2ll * siz[rx] * siz[ry];
    if( siz[rx] > siz[ry] ) swap( rx, ry );
    fa[rx] = ry, siz[ry] += siz[rx];
    return 1;
}
void active( int x ) { fa[x] = x; siz[x] = 1; }
bool check( int x ) { return fa[x] != 0; }
/* ======================================== */
} dsu;

struct State {
    int x; ll val;
} best[maxcnt];

bool better( State A, State B ) {
    if( A.x == 0 ) return 0;
    if( B.x == 0 ) return 1;
    i128 lhs = (i128)A.val * B.x;
    i128 rhs = (i128)B.val * A.x;
    if( lhs != rhs ) return lhs > rhs;
    return A.x > B.x;
}

int id( int x, int y ) {
    return better( best[y], best[x] ) ? y : x;
}

struct ST {
/* ======================================== */
int st[maxcnt][maxlg], lg2[maxn];
void init_lg( int N ) {
    lg2[1] = 0;
    for( int i = 2; i <= N; ++ i ) lg2[i] = lg2[i >> 1] + 1;
}
void init( int N ) {
    init_lg( N );
    for( int i = 1; i <= N; ++ i ) st[i][0] = i;
    for( int j = 1; ( 1 << j ) <= N; ++ j ) {
        for( int i = 1; i + ( 1 << j ) - 1 <= N; ++ i ) {
            st[i][j] = id( st[i][j - 1], st[i + ( 1 << (j - 1) )][j - 1] );
        }
    }
}
int query( int l, int r ) {
    int k = lg2[r - l + 1];
    return id( st[l][k], st[r - (1 << k) + 1][k] );
}
/* ======================================== */
} st;

struct Node {
    int val, idx;
    bool operator < ( const Node &oth ) const { return val < oth.val; }
} b[maxn];

void solve( ) {
    int n = read( ), T = read( );

    for( int i = 1; i <= n; ++ i ) {
        int x = read( );
        b[i] = { x, i };
    }

    sort( b + 1, b + n + 1 );
    dsu.init( n );

    ll val = 0;
    int cnt = 0;
    int mx = 0;

    for( int l = 1; l <= n; ) {
        int x = b[l].val;
        int r = l;

        while( r <= n && b[r].val == x ) {
            int pos = b[r].idx;
            
            dsu.active( pos );
            cnt ++, val ++;

            if( pos - 1 >= 1 && dsu.check( pos - 1 ) ) {
                if( dsu.merge( pos, pos - 1, val ) ) cnt --;
            }

            if( pos + 1 <= n && dsu.check( pos + 1 ) ) {
                if( dsu.merge( pos, pos + 1, val ) ) cnt --;
            }
            ++ r;
        }

        State cur = { x, val };
        if( better( cur, best[cnt] ) ) {
            best[cnt] = cur;
        }
        mx = max( mx, cnt );
        l = r;
    }

    st.init( mx );
    ll lst = 0;

    while( T -- ) {
        ll a = read( ), b = read( ), x = read( ), y = read( );
        int l = ((i128)a * lst + x - 1) % n + 1;
        int r = ((i128)b * lst + y - 1) % n + 1;
        if( l > r ) swap( l, r );
        ll lstmod = lst % n;

        if( l > min( r, mx ) ) {
            cout << "-1" << " " << "-1" << '\n';
            cout << l << " " << r << " " << lstmod << '\n';
            lst = 1;
            continue;
        }

        int idx = st.query( l, min( r, mx ) );
        State ans = best[idx];

        if( ans.x == 0 ) {
            cout << "-1" << " " << "-1" << '\n';
            cout << l << " " << r << " " << lstmod << '\n';
            lst = 1;
        }
        else {
            cout << ans.val << " " << ans.x << '\n';
            cout << l << " " << r << " " << lstmod << '\n';
            lst = ans.val * ans.x;
        }
    }
}
```

长度为 $n$ 的 $01$ 序列最多有 $\lceil n/2\rceil$ 个 $1$ 段，所以 ST 表第一维只需开到约 `5e5`。询问的 $r$ 可能大于实际出现过的最大段数 `mx`，查询前必须把右端点截到 `min(r,mx)`；区间为空或返回状态的 `x==0` 时无解。

强制在线计算 $l,r$ 时，输入系数与 `lastans` 的乘积也可能超过 `long long`，同样需要先转为 `__int128`。无解后题目规定 `lastans=1`；输出的 `lastans mod n` 是本次计算端点时使用的旧值。

预处理排序与并查集操作复杂度为 $O(n\log n)$，建立 ST 表为 $O(n\log n)$，每次询问为 $O(1)$。

```cpp title:"P5012" fold
struct DSU {
/* ======================================== */
int fa[maxn], siz[maxn];

void init( int N ) {
    for( int i = 1; i <= N; ++ i ) fa[i] = siz[i] = 0;
}

int find( int x ) {
    return fa[x] == x ? x : fa[x] = find( fa[x] );
}

void active( int x ) {
    fa[x] = x;
    siz[x] = 1;
}

bool check( int x ) {
    return fa[x] != 0;
}

bool merge( int x, int y, ll &val ) {
    int rx = find( x ), ry = find( y );
    if( rx == ry ) return false;

    val += 2LL * siz[rx] * siz[ry];
    if( siz[rx] > siz[ry] ) swap( rx, ry );
    fa[rx] = ry;
    siz[ry] += siz[rx];
    return true;
}
/* ======================================== */
} dsu;

struct State {
    int x;
    ll val;
} best[maxcnt];

bool better( State A, State B ) {
    if( A.x == 0 ) return false;
    if( B.x == 0 ) return true;

    i128 lhs = (i128)A.val * B.x;
    i128 rhs = (i128)B.val * A.x;
    if( lhs != rhs ) return lhs > rhs;
    return A.x > B.x;
}

int better_id( int x, int y ) {
    return better( best[y], best[x] ) ? y : x;
}

struct ST {
/* ======================================== */
int st[maxcnt][maxlg], lg2[maxcnt];

void init_lg( int N ) {
    lg2[1] = 0;
    for( int i = 2; i <= N; ++ i ) lg2[i] = lg2[i >> 1] + 1;
}

void init( int N ) {
    init_lg( N );
    for( int i = 1; i <= N; ++ i ) st[i][0] = i;

    for( int j = 1; ( 1 << j ) <= N; ++ j ) {
        for( int i = 1; i + ( 1 << j ) - 1 <= N; ++ i ) {
            st[i][j] = better_id(
                st[i][j - 1],
                st[i + ( 1 << ( j - 1 ) )][j - 1]
            );
        }
    }
}

int query( int l, int r ) {
    int k = lg2[r - l + 1];
    return better_id( st[l][k], st[r - ( 1 << k ) + 1][k] );
}
/* ======================================== */
} st;

struct Node {
    int val, idx;
    bool operator < ( const Node &other ) const {
        return val < other.val;
    }
} b[maxn];

void solve( ) {
    int n = read( ), T = read( );

    for( int i = 1; i <= n; ++ i ) {
        int x = read( );
        b[i] = { x, i };
    }

    sort( b + 1, b + n + 1 );
    dsu.init( n );

    ll val = 0;
    int cnt = 0, mx = 0;

    for( int l = 1; l <= n; ) {
        int x = b[l].val;
        int r = l;

        while( r <= n && b[r].val == x ) {
            int pos = b[r].idx;

            dsu.active( pos );
            ++ cnt;
            ++ val;

            if( pos > 1 && dsu.check( pos - 1 ) ) {
                if( dsu.merge( pos, pos - 1, val ) ) -- cnt;
            }
            if( pos < n && dsu.check( pos + 1 ) ) {
                if( dsu.merge( pos, pos + 1, val ) ) -- cnt;
            }

            ++ r;
        }

        State cur = { x, val };
        if( better( cur, best[cnt] ) ) best[cnt] = cur;
        mx = max( mx, cnt );
        l = r;
    }

    st.init( mx );
    ll lastans = 0;

    while( T -- ) {
        ll a = read( ), b = read( ), x = read( ), y = read( );
        ll lastmod = lastans % n;

        int l = ( (i128)a * lastans + x - 1 ) % n + 1;
        int r = ( (i128)b * lastans + y - 1 ) % n + 1;
        if( l > r ) swap( l, r );

        int qr = min( r, mx );
        if( l > qr ) {
            cout << "-1 -1\n";
            cout << l << ' ' << r << ' ' << lastmod << '\n';
            lastans = 1;
            continue;
        }

        State ans = best[st.query( l, qr )];

        if( ans.x == 0 ) {
            cout << "-1 -1\n";
            lastans = 1;
        }
        else {
            cout << ans.val << ' ' << ans.x << '\n';
            lastans = ans.val * ans.x;
        }

        cout << l << ' ' << r << ' ' << lastmod << '\n';
    }
}
```
