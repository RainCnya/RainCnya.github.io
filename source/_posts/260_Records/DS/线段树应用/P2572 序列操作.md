# P2572 序列操作

> [!question] [P2572 [SCOI2010] 序列操作](https://www.luogu.com.cn/problem/P2572)
> 维护一个 01 序列，支持区间全部改为零、全部改为一、整体翻转，并询问区间内一的数量或最长连续一段。
>
> 数据规模：$n,m\le 10^5$，题目下标从零开始。

这题其实像前面几题的组装版：[[浅谈线段树懒标记|P2574 的区间翻转]]、[[浅谈线段树懒标记|P1253 的标记复合]]，以及 [P6492 STEP](https://www.luogu.com.cn/problem/P6492) 的信息合并。

回到本题，区间和只能回答一的数量，要询问最长连续一，还得同时维护 $0$ 和 $1$ 各自的最长前缀、后缀与区间答案。

不然由于区间会翻转或者覆盖，没法直接维护前缀 1 和 后缀 1 的数量，所以还需要跟 前缀 0 和 后缀 0 打配合。

然后就是类似的合并了，主要是写起来很麻烦。

覆盖 0 或 1 时重置翻转；翻转时交换 0/1 的全部统计，已有覆盖就翻转覆盖值，否则异或累计翻转，下传顺序保持先覆盖、后翻转。查询最长一段时应直接取返回节点的 `mx[1]`。

一道很综合的线段树题，这才青题吗？

```cpp title:"P2572" fold
struct SegTree {
/* ================================================== */
struct Node { 
    int l, r; 
    int sum, cov, rev;
    int lmx[2], rmx[2], mx[2];
} tr[maxn << 2];

int len( int u ) { 
    return tr[u].r - tr[u].l + 1; 
}

void push_up( int u ) {
    tr[u].sum = tr[ls].sum + tr[rs].sum;

    for( int c = 0; c <= 1; ++ c ) {
        tr[u].lmx[c] = tr[ls].lmx[c];
        if( tr[ls].lmx[c] == len( ls ) ) {
            tr[u].lmx[c] += tr[rs].lmx[c];
        }
        tr[u].rmx[c] = tr[rs].rmx[c];
        if( tr[rs].rmx[c] == len( rs ) ) {
            tr[u].rmx[c] += tr[ls].rmx[c];
        }
        tr[u].mx[c] = max({ 
            tr[ls].mx[c], tr[rs].mx[c], 
            tr[ls].rmx[c] + tr[rs].lmx[c] 
        });
    }
}

void apply_rev( int u ) {
    tr[u].sum = len( u ) - tr[u].sum;
    swap( tr[u].lmx[0], tr[u].lmx[1] );
    swap( tr[u].rmx[0], tr[u].rmx[1] );
    swap( tr[u].mx[0], tr[u].mx[1] );
    if( tr[u].cov != -1 ) tr[u].cov ^= 1;
    else tr[u].rev ^= 1;
}

void apply_cov( int u, int val ) {
    tr[u].sum = val ? len( u ) : 0;
    tr[u].lmx[0] = tr[u].rmx[0] = tr[u].mx[0] = val == 0 ? len( u ) : 0;
    tr[u].lmx[1] = tr[u].rmx[1] = tr[u].mx[1] = val == 1 ? len( u ) : 0;
    tr[u].cov = val;
    tr[u].rev = 0;
}

void push_down( int u ) {
    if( tr[u].cov != -1 ) {
        apply_cov( ls, tr[u].cov );
        apply_cov( rs, tr[u].cov );
        tr[u].cov = -1;
    }
    if( tr[u].rev ) {
        apply_rev( ls );
        apply_rev( rs );
        tr[u].rev = 0;
    }
}

void build( int u, int l, int r, ll val[] ) {
    tr[u] = { l, r, 0, -1, 0 };
    if( l == r ) {
        int c = val[l];
        int d = c ^ 1;
        tr[u].sum = c;
        tr[u].lmx[c] = tr[u].rmx[c] = tr[u].mx[c] = 1;
        tr[u].lmx[d] = tr[u].rmx[d] = tr[u].mx[d] = 0;
        return ;
    }
    int mid = ( l + r ) >> 1;
    build( ls, l, mid, val );
    build( rs, mid + 1, r, val );
    push_up( u );
}

void upd( int u, int l, int r, int opt ) {
    if( l <= tr[u].l && tr[u].r <= r ) {
        if( opt == 0 || opt == 1 ) apply_cov( u, opt );
        else apply_rev( u );
        return ;
    }
    push_down( u );
    int mid = ( tr[u].l + tr[u].r ) >> 1;
    if( l <= mid ) upd( ls, l, r, opt );
    if( mid < r ) upd( rs, l, r, opt );
    push_up( u );
}

Node merge( Node L, Node R ) {
    Node res = { L.l, R.r, L.sum + R.sum, -1, 0 };
    int lenL = L.r - L.l + 1;
    int lenR = R.r - R.l + 1;
    for( int c = 0; c <= 1; ++ c ) {
        res.lmx[c] = L.lmx[c];
        if( L.lmx[c] == lenL ) {
            res.lmx[c] += R.lmx[c];
        }
        res.rmx[c] = R.rmx[c];
        if( R.rmx[c] == lenR ) {
            res.rmx[c] += L.rmx[c];
        }
        res.mx[c] = max({ 
            L.mx[c], R.mx[c], 
            L.rmx[c] + R.lmx[c] 
        });
    }
    return res;
}

Node ask( int u, int l, int r ) {
    if( l <= tr[u].l && tr[u].r <= r ) return tr[u];
    push_down( u );
    int mid = ( tr[u].l + tr[u].r ) >> 1;
    if( r <= mid ) return ask( ls, l, r );
    if( mid < l ) return ask( rs, l, r );
    return merge( ask( ls, l, r ), ask( rs, l, r ) );
}
/* ================================================== */
} seg;

ll a[maxn];

void solve( ) {
    int n = read( ), m = read( );
    for( int i = 1; i <= n; ++ i ) a[i] = read( );
    seg.build( 1, 1, n, a );

    for( int i = 1; i <= m; ++ i ) {
        int opt = read( );
        int l = read( ) + 1, r = read( ) + 1;
        if( opt == 0 || opt == 1 || opt == 2 ) {
            seg.upd( 1, l, r, opt );
        }
        else if( opt == 3 ) {
            cout << seg.ask( 1, l, r ).sum << '\n';
        }
        else {
            cout << seg.ask( 1, l, r ).mx[1] << '\n';
        }
    }
}
```
